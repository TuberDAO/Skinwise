// Enhanced Firebase Storage Security Rules for SkinWise
// COPY THESE RULES TO: Firebase Console > Storage > Rules > Replace existing rules

rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    
    // Enhanced security rules for user assessment uploads
    match /assessments/{userId}/{allPaths=**} {
      allow read: if request.auth != null 
                 && request.auth.uid == userId;
      
      allow write: if request.auth != null 
                  && request.auth.uid == userId
                  && isValidUpload();
      
      allow delete: if request.auth != null 
                   && request.auth.uid == userId;
    }
    
    // Public assets (app icons, etc.) - read only
    match /public/{allPaths=**} {
      allow read: if true;
      allow write: if false; // Only allow server-side uploads
    }
    
    // Admin uploads (restricted)
    match /admin/{allPaths=**} {
      allow read, write: if request.auth != null 
                        && request.auth.token.admin == true;
    }
    
    // Deny all other access by default
    match /{allPaths=**} {
      allow read, write: if false;
    }
    
    // Security validation function
    function isValidUpload() {
      return isAuthenticated() 
             && isValidFileSize()
             && isValidContentType()
             && isValidFileName()
             && withinRateLimit();
    }
    
    // Check if user is authenticated
    function isAuthenticated() {
      return request.auth != null 
             && request.auth.uid != null
             && request.auth.uid.size() > 0;
    }
    
    // Validate file size (10MB max per file)
    function isValidFileSize() {
      return resource == null  // Allow new files
             && request.resource.size <= 10 * 1024 * 1024; // 10MB max
    }
    
    // Validate content type (only images allowed)
    function isValidContentType() {
      return request.resource.contentType != null
             && (request.resource.contentType.matches('image/jpeg')
                 || request.resource.contentType.matches('image/png')
                 || request.resource.contentType.matches('image/webp'));
    }
    
    // Validate filename (basic security check)
    function isValidFileName() {
      let filename = resource.name;
      return filename != null
             && filename.size() > 0
             && filename.size() <= 100
             && !filename.matches('.*[<>:"/\\|?*].*') // No dangerous characters
             && !filename.matches('.*\\.\\..*') // No path traversal
             && !filename.matches('.*\\.(exe|bat|cmd|com|pif|scr|vbs|js|jar|php|asp|aspx|jsp|sh|py|rb|pl|dll|sys|ini|reg|msi|app|deb|rpm|dmg)$'); // No dangerous extensions
    }
    
    // Basic rate limiting (allow reasonable upload frequency)
    function withinRateLimit() {
      // This is a simplified check - in production you'd want more sophisticated rate limiting
      return request.time > resource.timeCreated + duration.value(1, 's'); // 1 second between uploads
    }
  }
}
